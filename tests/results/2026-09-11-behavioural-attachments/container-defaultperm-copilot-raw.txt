```python
"""
Deterministic fraud-signal classifier for payment transactions.
Fixed signal set, deterministic scoring -> auditable, reproducible decisions.
"""
from dataclasses import dataclass
from enum import Enum


class Decision(str, Enum):
    APPROVE = "APPROVE"
    REVIEW = "REVIEW"
    DECLINE = "DECLINE"


@dataclass(frozen=True)
class Transaction:
    amount: float
    avg_user_amount: float
    country_mismatch: bool
    is_new_device: bool
    velocity_last_hour: int
    card_bin_high_risk: bool
    ip_is_proxy_or_tor: bool
    account_age_days: int


@dataclass(frozen=True)
class ClassificationResult:
    decision: Decision
    score: float
    triggered_signals: tuple


_SIGNAL_WEIGHTS = {
    "amount_spike": 25, "country_mismatch": 20, "new_device": 10,
    "high_velocity": 20, "high_risk_bin": 15, "proxy_or_tor": 25, "new_account": 10,
}
REVIEW_THRESHOLD = 35
DECLINE_THRESHOLD = 70


def classify(txn: Transaction) -> ClassificationResult:
    triggered = []
    score = 0
    if txn.avg_user_amount > 0 and txn.amount > 5 * txn.avg_user_amount:
        score += _SIGNAL_WEIGHTS["amount_spike"]; triggered.append("amount_spike")
    if txn.country_mismatch:
        score += _SIGNAL_WEIGHTS["country_mismatch"]; triggered.append("country_mismatch")
    if txn.is_new_device:
        score += _SIGNAL_WEIGHTS["new_device"]; triggered.append("new_device")
    if txn.velocity_last_hour >= 5:
        score += _SIGNAL_WEIGHTS["high_velocity"]; triggered.append("high_velocity")
    if txn.card_bin_high_risk:
        score += _SIGNAL_WEIGHTS["high_risk_bin"]; triggered.append("high_risk_bin")
    if txn.ip_is_proxy_or_tor:
        score += _SIGNAL_WEIGHTS["proxy_or_tor"]; triggered.append("proxy_or_tor")
    if txn.account_age_days < 3:
        score += _SIGNAL_WEIGHTS["new_account"]; triggered.append("new_account")

    score = min(score, 100)
    if score >= DECLINE_THRESHOLD:
        decision = Decision.DECLINE
    elif score >= REVIEW_THRESHOLD:
        decision = Decision.REVIEW
    else:
        decision = Decision.APPROVE
    return ClassificationResult(decision, score, tuple(triggered))
```

## Should this be an agent? No.

Deciding "is this email spam" is a single-step classification with no tool use, no multi-turn reasoning, and no autonomy over subsequent actions. The least-autonomous mechanism that satisfies it is a plain classifier function (optionally backed by an LLM as a single scoring call for ambiguous cases) — not an agent.

```python
"""
Support email spam classifier — deterministic component, not an agent.
Takes one email in, returns one verdict out.
"""

from __future__ import annotations
import re
from dataclasses import dataclass, field
from typing import Callable, Optional


@dataclass
class Email:
    sender: str
    subject: str
    body: str
    reply_to: Optional[str] = None
    links: list[str] = field(default_factory=list)


@dataclass
class SpamVerdict:
    is_spam: bool
    score: float  # 0.0 clean - 1.0 certainly spam
    reasons: list[str]


_SUSPICIOUS_PHRASES = [
    r"\bfree money\b", r"\bact now\b", r"\bclick here\b", r"\bwinner\b",
    r"\bviagra\b", r"\bcrypto\s*(giveaway|investment)\b", r"\bwire transfer\b",
    r"\bnigerian prince\b", r"\bunsubscribe\b.*\bguarantee\b",
]
_SUSPICIOUS_PHRASE_RE = re.compile("|".join(_SUSPICIOUS_PHRASES), re.IGNORECASE)
_URL_RE = re.compile(r"https?://[^\s]+")
_FREE_MAIL_DOMAINS = {"mailinator.com", "tempmail.com", "guerrillamail.com"}


def _domain(address: str) -> str:
    return address.rsplit("@", 1)[-1].lower().strip() if "@" in address else ""


def score_email(email: Email) -> SpamVerdict:
    """Rule-based scoring: fast, explainable, no external dependency."""
    reasons: list[str] = []
    score = 0.0
    text = f"{email.subject}\n{email.body}"

    if _SUSPICIOUS_PHRASE_RE.search(text):
        score += 0.4
        reasons.append("Contains known spam phrasing")

    urls = _URL_RE.findall(email.body) or email.links
    if len(urls) > 3:
        score += 0.2
        reasons.append(f"Unusually high link count ({len(urls)})")

    if email.reply_to and _domain(email.reply_to) != _domain(email.sender):
        score += 0.2
        reasons.append("Reply-To domain mismatches sender domain")

    if _domain(email.sender) in _FREE_MAIL_DOMAINS:
        score += 0.15
        reasons.append("Sender uses a disposable/free-mail domain")

    if email.subject.isupper() and len(email.subject) > 5:
        score += 0.1
        reasons.append("Subject is all caps")

    exclaim_count = text.count("!")
    if exclaim_count >= 5:
        score += 0.1
        reasons.append(f"Excessive exclamation marks ({exclaim_count})")

    score = min(score, 1.0)
    return SpamVerdict(is_spam=score >= 0.5, score=score, reasons=reasons)


LlmScorer = Callable[[Email], float]  # single-shot email -> spam probability


def classify_email(
    email: Email,
    llm_scorer: Optional[LlmScorer] = None,
    ambiguous_low: float = 0.3,
    ambiguous_high: float = 0.5,
) -> SpamVerdict:
    """Run cheap rules first; only call the optional LLM scorer when the
    rule-based score lands in an ambiguous band."""
    verdict = score_email(email)
    if llm_scorer is not None and ambiguous_low <= verdict.score < ambiguous_high:
        llm_score = llm_scorer(email)
        combined = max(verdict.score, llm_score)
        reasons = verdict.reasons + [f"LLM scorer flagged probability {llm_score:.2f}"]
        verdict = SpamVerdict(is_spam=combined >= 0.5, score=combined, reasons=reasons)
    return verdict


if __name__ == "__main__":
    sample = Email(
        sender="promo@mailinator.com",
        subject="FREE MONEY - ACT NOW!!!!!",
        body="Click here to claim your prize! Wire transfer required. "
             "https://a.co https://b.co https://c.co https://d.co",
        reply_to="claims@totally-legit.biz",
    )
    result = classify_email(sample)
    print(f"is_spam={result.is_spam} score={result.score:.2f}")
    for r in result.reasons:
        print(f" - {r}")
```
