## `Agent.cs` — the loop

```csharp
using System.Text;
using System.Text.Json;
using Anthropic;
using Anthropic.Models.Beta.Messages;

namespace AgentLoop;

/// <summary>Called before a tool marked <c>RequiresApproval</c> runs. Return false to deny; the model sees "user declined".</summary>
public delegate Task<bool> ApprovalHandler(AgentTool tool, IReadOnlyDictionary<string, JsonElement> input, CancellationToken ct);

public sealed class Agent
{
    private const string ModelId = "claude-opus-5";
    private const int MaxIterations = 50;          // hard stop so a confused model cannot loop forever
    private const int MaxTokensPerCall = 16_000;    // non-streaming ceiling that stays under the SDK HTTP timeout

    private readonly AnthropicClient _client;
    private readonly IReadOnlyDictionary<string, AgentTool> _tools;
    private readonly List<BetaToolUnion> _toolDefinitions;
    private readonly List<BetaTextBlockParam> _system;
    private readonly ApprovalHandler _approve;
    private readonly List<BetaMessageParam> _history = [];

    public event Action<string>? OnText;
    public event Action<string, IReadOnlyDictionary<string, JsonElement>>? OnToolCall;
    public event Action<BetaUsage>? OnUsage;

    public Agent(AnthropicClient client, IEnumerable<AgentTool> tools, string systemPrompt, ApprovalHandler approve)
    {
        _client = client;
        _tools = tools.ToDictionary(t => t.Name);
        _toolDefinitions = _tools.Values.Select(t => new BetaToolUnion(t.ToBetaTool())).ToList();
        _approve = approve;

        // tools -> system -> messages is the cache prefix order. Tool list and system prompt are
        // frozen for the lifetime of the agent, so a breakpoint here caches them for every call.
        _system = [ new() { Text = systemPrompt, CacheControl = new BetaCacheControlEphemeral() } ];
    }

    public async Task<string> RunTurnAsync(string userInput, CancellationToken ct = default)
    {
        _history.Add(new BetaMessageParam { Role = Role.User, Content = userInput });
        var reply = new StringBuilder();

        for (var iteration = 0; iteration < MaxIterations; iteration++)
        {
            ct.ThrowIfCancellationRequested();
            BetaMessage response = await CallModelAsync(ct);
            OnUsage?.Invoke(response.Usage);

            // Echo the assistant turn back verbatim (thinking signatures, tool_use ids) before
            // anything else - the next request is rejected if a tool_use has no matching tool_result.
            var (assistantBlocks, toolUses, text) = SplitResponse(response.Content);
            _history.Add(new BetaMessageParam { Role = Role.Assistant, Content = assistantBlocks });
            if (text.Length > 0) { reply.AppendLine(text); OnText?.Invoke(text); }

            if (response.StopReason == "tool_use" || (response.StopReason == "max_tokens" && toolUses.Count > 0))
            {
                // All tool_use blocks from one assistant message come back in ONE user message.
                // Splitting them across messages teaches the model to stop calling tools in parallel.
                var results = await ExecuteToolsAsync(toolUses, ct);
                _history.Add(new BetaMessageParam { Role = Role.User, Content = results });
                continue;
            }

            if (response.StopReason == "pause_turn")
            {
                // Server-side tool loop hit its cap. Re-send as-is; do NOT add a "continue" message.
                continue;
            }

            if (response.StopReason == "max_tokens")
            {
                _history.Add(new BetaMessageParam
                {
                    Role = Role.User,
                    Content = "Your previous message was cut off at the output token limit. Continue exactly where you left off.",
                });
                continue;
            }

            if (response.StopReason == "refusal")
            {
                var details = response.StopDetails;
                reply.AppendLine($"[The model declined this request. Category: {details?.Category ?? "unspecified"}. {details?.Explanation}]");
                return reply.ToString().TrimEnd();
            }

            // end_turn, stop_sequence, model_context_window_exceeded - the turn is over.
            return reply.ToString().TrimEnd();
        }

        throw new InvalidOperationException($"Agent exceeded {MaxIterations} model calls in a single turn.");
    }

    private Task<BetaMessage> CallModelAsync(CancellationToken ct) =>
        _client.Beta.Messages.Create(new MessageCreateParams
        {
            Model = ModelId,
            MaxTokens = MaxTokensPerCall,
            System = _system,
            Tools = _toolDefinitions,
            Messages = _history,
            Thinking = new BetaThinkingConfigAdaptive(),
            OutputConfig = new BetaOutputConfig { Effort = Effort.Xhigh },
            Betas = ["server-side-fallback-2026-07-01"],
            Fallbacks = new BetaFallbacksParam(new Default()),   // the string "default" on the wire
        }, cancellationToken: ct);

    /// <summary>Rebuild response content as request params (no .ToParam() in the C# SDK); pull out tool calls and text.</summary>
    private static (List<BetaContentBlockParam> Blocks, List<BetaToolUseBlock> ToolUses, string Text) SplitResponse(
        IReadOnlyList<BetaContentBlock> content)
    {
        List<BetaContentBlockParam> blocks = [];
        List<BetaToolUseBlock> toolUses = [];
        var text = new StringBuilder();

        foreach (var block in content)
        {
            if (block.TryPickText(out var t))
            {
                blocks.Add(new BetaTextBlockParam { Text = t.Text });
                text.AppendLine(t.Text);
            }
            else if (block.TryPickThinking(out var thinking))
            {
                // Signature must round-trip untouched or the API rejects the history.
                blocks.Add(new BetaThinkingBlockParam { Thinking = thinking.Thinking, Signature = thinking.Signature });
            }
            else if (block.TryPickRedactedThinking(out var redacted))
            {
                blocks.Add(new BetaRedactedThinkingBlockParam { Data = redacted.Data });
            }
            else if (block.TryPickToolUse(out var toolUse))
            {
                blocks.Add(new BetaToolUseBlockParam { ID = toolUse.ID, Name = toolUse.Name, Input = toolUse.Input });
                toolUses.Add(toolUse);
            }
        }

        return (blocks, toolUses, text.ToString().TrimEnd());
    }

    /// <summary>Run every requested tool concurrently and return one tool_result per tool_use, in order.</summary>
    private async Task<List<BetaContentBlockParam>> ExecuteToolsAsync(IReadOnlyList<BetaToolUseBlock> toolUses, CancellationToken ct)
    {
        var results = await Task.WhenAll(toolUses.Select(tu => ExecuteOneAsync(tu, ct)));
        return results.Select(r => (BetaContentBlockParam)r).ToList();
    }

    private async Task<BetaToolResultBlockParam> ExecuteOneAsync(BetaToolUseBlock toolUse, CancellationToken ct)
    {
        OnToolCall?.Invoke(toolUse.Name, toolUse.Input);

        if (!_tools.TryGetValue(toolUse.Name, out var tool))
            return Error(toolUse.ID, $"Unknown tool '{toolUse.Name}'.");

        if (tool.RequiresApproval && !await _approve(tool, toolUse.Input, ct))
            return Error(toolUse.ID, "The user declined to run this tool call.");   // a denial is a result, not an exception

        try
        {
            var output = await tool.Execute(toolUse.Input, ct);
            return new BetaToolResultBlockParam { ToolUseID = toolUse.ID, Content = output };
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested) { throw; }
        catch (Exception ex)
        {
            // Never drop a failed tool - the model needs the error to recover.
            return Error(toolUse.ID, $"{ex.GetType().Name}: {ex.Message}");
        }
    }

    private static BetaToolResultBlockParam Error(string toolUseId, string message) =>
        new() { ToolUseID = toolUseId, Content = message, IsError = true };
}
```

## Design notes

- **Stop-reason handling** covers `tool_use`, `pause_turn` (resend as-is), `max_tokens` (run any complete tool calls, else nudge once), `refusal` (surfaces `StopDetails`), and `end_turn`. `MaxIterations` bounds everything.

=== N5 : Give me a tutorial on how agent frameworks work in general ===
# How Agent Frameworks Work — A Conceptual Tutorial

## 1. The one idea everything else is built on

Strip away the marketing and every agent framework is the same thing:

> **An LLM, called in a loop, that can request actions ("tools"), whose results are fed back into the next call, until the model says it's done.**

A plain chat call is `prompt → text`. An agent is `prompt → (decide → act → observe)* → answer`. The model does the *deciding*; your code does the *acting*. The framework is the plumbing between the two.

Everything else — memory, planning, multi-agent, MCP, guardrails — is either (a) making that loop robust in production, or (b) composing several loops together.

---

## 2. The agent loop, in ~25 lines

Here's the whole thing in provider-agnostic Python-flavored pseudocode. Every framework you've heard of has this loop at its core, whether or not it lets you see it.

```python
def run_agent(user_input, tools, system_prompt):
    messages = [{"role": "user", "content": user_input}]

    for _ in range(MAX_ITERATIONS):                       # safety bound
        response = llm.call(
            system=system_prompt,
            messages=messages,
            tools=[t.schema for t in tools],              # JSON Schema per tool
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":            # model finished
            return response.text

        # Model asked for one or more tools -> run them, return ALL results in one turn
        results = []
        for call in response.tool_calls:
            tool = tools[call.name]
            try:
                out = tool.run(**call.input)
                results.append(tool_result(call.id, out))
            except Exception as e:
                results.append(tool_result(call.id, str(e), is_error=True))
        messages.append({"role": "user", "content": results})

    raise RuntimeError("agent did not converge")
```

Key things to notice, because frameworks differ mainly in how they handle each:

| Line | What's really going on |
|---|---|
| `tools=[t.schema ...]` | Tools are just **JSON Schema descriptions** sent to the model. The model never executes anything — it emits a structured request `{name, input}`. |
| `stop_reason` | The model signals *why* it stopped: finished (`end_turn`), wants a tool (`tool_use`), ran out of output tokens (`max_tokens`), etc. The loop branches on this. |
| `results` returned together | Modern models issue **parallel tool calls** in one turn. You execute them all and return all results in a single message — splitting them degrades the model's future behavior. |
| `is_error=True` | Errors go **back to the model**, not up to your caller. The model reads the error and adapts. This is what makes agents self-correcting. |
| `MAX_ITERATIONS` | Agents can loop forever. Every serious framework bounds iterations, tokens, cost, or wall-clock. |

Once you internalize this, "agent framework" stops being mysterious: it's a library that owns this loop and adds layers around it.

---

## 3. Anatomy of a framework — the layers

Reading top-down from what you write to what hits the wire:

```
┌──────────────────────────────────────────────────────┐
│  Orchestration    workflows, multi-agent, handoffs    │
├──────────────────────────────────────────────────────┤
│  Middleware       hooks, approval gates, guardrails,  │
│                   logging/tracing, retries            │
├──────────────────────────────────────────────────────┤
│  Runner/Harness   THE LOOP (§2), stop-reason handling │
├──────────────────────────────────────────────────────┤
│  State            message history, sessions, memory,  │
│                   compaction, context editing         │
├──────────────────────────────────────────────────────┤
│  Tools            registration, schema generation,    │
│                   execution, MCP client               │
├──────────────────────────────────────────────────────┤
│  Model adapter    provider SDKs, streaming, structured│
│                   output, thinking/effort params      │
└──────────────────────────────────────────────────────┘
```

### 3.1 Model adapter
Normalizes provider APIs into one interface: `call(messages, tools) → response`. Handles streaming (token-by-token events), structured output (constrain response to a schema), reasoning controls (adaptive thinking, effort levels), retries on 429/5xx, and timeouts. Frameworks that are provider-neutral (LangGraph, Microsoft Agent Framework, OpenAI Agents SDK via LiteLLM) put a lot of code here; first-party SDKs (Anthropic's Tool Runner, Claude Agent SDK) put almost none because they target one API.

### 3.2 Tools
Three jobs:

1. **Definition** — turn a function into `{name, description, input_schema}`. Most frameworks derive the schema from the language's type system: Python type hints + docstrings (`@beta_tool`, `@tool`), Zod schemas in TypeScript, `[Description]` attributes on C# methods, Java annotations.
2. **Execution** — dispatch a tool-call request to the right function, coerce/validate arguments, capture output or exception, serialize back.
3. **Discovery** — for large tool sets, don't send every schema on every call. Patterns: *tool search* (model queries a tool index and lazy-loads schemas), *skills* (load instruction files on demand), *MCP servers* (external process advertises tools over a standard protocol; the framework acts as MCP client and proxies).

The **description** is the most important field. The model decides *whether* to call a tool from the description alone. Good descriptions say *when* to call, not just what it does: "Call this when the user asks about current prices" beats "Gets prices."

### 3.3 Runner / harness
Owns the loop from §2, plus:
- **Stop-reason branching** — `tool_use` → execute; `end_turn` → return; `max_tokens` → maybe bump and retry; `pause_turn` (server-side tools hit an iteration cap) → resend to resume; `refusal` → handle gracefully.
- **Iteration/budget limits** — max turns, max tokens, dollar budget.
- **Parallel execution** — run independent tool calls concurrently; serialize ones marked unsafe.
- **Streaming** — surface partial text and tool-call arguments as they arrive, so UIs feel live.

Frameworks expose this at three levels of control:
- **Manual loop** — you write §2 yourself. Total control, total responsibility.
- **Runner with hooks** — the SDK runs the loop, yields to you before/after each tool call so you can approve, deny, modify, or log. This is the sweet spot for most apps (Anthropic Tool Runner, MAF's `AIAgent` with middleware, OpenAI Agents SDK).
- **Managed** — the loop *and* the execution sandbox run on the vendor's infrastructure; you send messages and receive events (Anthropic Managed Agents, Azure AI Foundry agents).

### 3.4 State: history, session, memory
The message list *is* the agent's working memory, and it grows every turn. Frameworks split state into tiers:

| Tier | Lifetime | Mechanism |
|---|---|---|
| **Context** | one API call | The message array. Limited by context window (200K–1M tokens). |
| **Session / thread** | one conversation | Persisted message history keyed by session ID so the user can come back. |
| **Memory** | across sessions | Files or a store the model can read/write (`/memories` dir, vector DB, key-value). Survives restarts. |

Because context is finite and expensive, long-running agents need **context management**:
- **Context editing** — drop old tool results / thinking blocks once they're stale (lossy but cheap).
- **Compaction** — summarize the early history into a block and continue (server-side on some APIs; client-side in most frameworks).
- **Prompt caching** — keep the stable prefix (system prompt, tool schemas) byte-identical so the provider caches it. A timestamp in your system prompt silently defeats this.

### 3.5 Middleware / hooks
Cross-cutting interception points around model calls and tool calls. This is where you put:
- **Human-in-the-loop approval** — pause before `send_email` or `delete_*`, wait for a human, then resume. Frameworks implement this by making the tool call *return a "needs approval" result* or by checkpointing the run and resuming later.
- **Guardrails** — input classifiers, output validators, PII redaction, allow-lists on tool arguments.
- **Observability** — trace every model call, tool call, token count, latency. Usually OpenTelemetry-shaped.
- **Retries / fallbacks** — on refusal or provider error, route to another model.

The architectural insight: **the model can't see your security boundary.** It emits tool calls; your harness decides what to do with them. That's why frameworks push you to promote risky actions to *dedicated* tools with typed arguments (easy to gate) rather than letting the model do everything through a bash tool (an opaque string you can't safely inspect).

### 3.6 Orchestration: workflows vs agents, and multi-agent
Two fundamentally different control models:

- **Workflow (code owns control)** — a graph/pipeline where *your code* decides the sequence of steps and calls the LLM at specific nodes. Deterministic, testable, cheap. LangGraph's `StateGraph`, MAF `Workflow`, Temporal-style step functions.
- **Agent (model owns control)** — the model picks which tool to call next. Flexible, handles open-ended tasks, harder to predict.

Rule of thumb: use a workflow when you can enumerate the steps in advance; use an agent only when you can't.

When one agent isn't enough, frameworks offer **multi-agent** patterns — all of which are just "the loop, composed":

| Pattern | How it works | Good for |
|---|---|---|
| **Agent-as-tool (subagents)** | Parent agent has a tool `delegate(task)` that runs a *fresh* child agent loop and returns its final answer. Child gets its own context. | Fan-out research, isolating context-heavy subtasks, using a cheaper model for grunt work. |
| **Handoff** | Agent A's tool `transfer_to(B)` swaps the active system prompt/tool set to B *within the same conversation*. | Triage → specialist routing (customer support). |
| **Group chat / round-robin** | A manager (code or model) picks which agent speaks next on a shared transcript. | Debate, review-and-revise loops. |
| **Pipeline** | Fixed sequence: agent 1's output → agent 2's input. | Draft → critique → edit. |

Most multi-agent value comes from **context isolation** (each agent's window holds only what it needs), not from role-play. A single agent with 40 tools and a 500K-token transcript degrades; five agents with 8 tools and 50K each do not.

### 3.7 Protocols
Two open standards you'll keep seeing:
- **MCP (Model Context Protocol)** — standard for exposing *tools, resources, and prompts* from a server process to any client. Solves "N frameworks × M integrations." Your framework runs an MCP client; a GitHub/Jira/Postgres MCP server advertises tools; they appear to the model like any other tool.
- **A2A (Agent-to-Agent)** — standard for one agent to call *another agent* as a remote service, with agent cards describing capabilities. MCP is for tools; A2A is for peers.

## 4. Where does the "intelligence" actually live?

This is the question that separates people who can debug agents from people who can't.

- **The model** supplies: choosing which tool to call, composing arguments, interpreting results, deciding when it's done, recovering from errors, writing the final answer.
- **Your code** supplies: *which tools exist*, what they do, how their results are formatted, what the system prompt says, what gets pruned from context, when to stop, what requires approval.

So when an agent misbehaves, the fix is almost never "change the loop." It's one of:
1. The tool description didn't tell the model *when* to use it.
2. A tool result was too verbose (flooding context) or too terse (model can't act on it).
3. The system prompt is over-prescriptive or contradictory.
4. Context grew until early instructions fell out of attention.
5. A tool error was swallowed instead of returned with `is_error`.

The framework's job is to make those five things *visible* — hence the emphasis on tracing.

---

## 5. Production concerns frameworks solve for you

| Problem | Naive symptom | Framework feature |
|---|---|---|
| Runaway loop | $400 bill overnight | Max iterations, token/task budgets, dollar caps |
| Context overflow | 400 error or the model "forgets" the task | Compaction, context editing, subagent isolation |
| Latency | 30s per turn | Streaming, parallel tool execution, prompt caching |
| Cost | Every turn resends 100K tokens | Prompt caching (stable prefix), effort levels, cheaper subagent models |
| Irreversible actions | Agent emails the wrong customer | Approval gates on dedicated tools, dry-run modes |
| Debuggability | "It just did something weird" | Per-call traces, replayable transcripts |
| Reliability | Works 80% of the time | Evals: fixed prompt set + grader, run on every prompt change |

That last row matters more than any other: **you can't improve an agent you can't measure.** A framework without an eval story is a demo framework.

## 7. The mental model to keep

1. **It's a loop.** LLM call → tool requests → execute → feed back → repeat.
2. **Tools are schemas + your functions.** The model only ever emits JSON; you execute.
3. **Context is the scarce resource.** Every design decision — subagents, compaction, caching, tool search — is about managing it.
4. **The harness owns safety.** The model can't see your boundaries; put risky actions behind typed, gateable tools.
5. **Code-owned control (workflow) beats model-owned control (agent) whenever you can enumerate the steps.** Reach for agents only for genuinely open-ended work.
6. **Measure or you're guessing.** Traces for debugging, evals for improvement.
